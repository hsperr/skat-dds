cmake_minimum_required(VERSION 3.15)
project(skatdds LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(SKATDDS_NATIVE "Optimize for the build machine's CPU (-march=native)" ON)

find_package(Threads REQUIRED)
add_library(skatdds SHARED src/skatdds.cpp)
target_include_directories(skatdds PUBLIC include)
target_link_libraries(skatdds PRIVATE Threads::Threads)
target_compile_options(skatdds PRIVATE -O3 -Wall -Wextra)
if(SKATDDS_NATIVE)
  include(CheckCXXCompilerFlag)
  check_cxx_compiler_flag(-march=native HAS_MARCH_NATIVE)
  if(HAS_MARCH_NATIVE)
    target_compile_options(skatdds PRIVATE -march=native)
  else()
    check_cxx_compiler_flag(-mcpu=native HAS_MCPU_NATIVE)
    if(HAS_MCPU_NATIVE)
      target_compile_options(skatdds PRIVATE -mcpu=native)
    endif()
  endif()
endif()

add_executable(skatdds_example examples/example.c)
target_link_libraries(skatdds_example PRIVATE skatdds)
